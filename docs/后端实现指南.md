#### JWT Token

```go
type Claims struct {
  ID       int    `json:"id"`
  Username string `json:"username"`
  jwt.StandardClaims
}

func GenerateToken(user *User) (string, error) {
  claims := &Claims{
    ID:       user.ID,
    Username: user.Username,
    StandardClaims: jwt.StandardClaims{
      ExpiresAt: time.Now().Add(24 * time.Hour).Unix(),
      IssuedAt:  time.Now().Unix(),
      Issuer:    "gomind",
    },
  }
  
  token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
  tokenString, err := token.SignedString([]byte("your-secret-key"))
  return tokenString, err
}
```

#### 用户注册

```go
func Register(username, email, password, captcha string) error {
  // 验证验证码
  if !verifyCaptcha(email, captcha) {
    return errors.New("验证码错误或已过期")
  }
  
  // 检查邮箱是否已存在
  var user User
  if err := db.Where("email = ?", email).First(&user).Error; err == nil {
    return errors.New("邮箱已被注册")
  }
  
  // 密码哈希
  hashedPassword, _ := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
  
  // 创建用户
  newUser := User{
    Username:     username,
    Email:        email,
    PasswordHash: string(hashedPassword),
  }
  
  return db.Create(&newUser).Error
}
```

### 3. 帖子操作

#### 创建帖子

```go
func CreatePost(userID int, content string, images []string) error {
  post := Post{
    ID:      uuid.New().String(),
    UserId:  userID,
    Content: content,
    Images:  images,
  }
  
  // 事务处理
  return db.Transaction(func(tx *gorm.DB) error {
    if err := tx.Create(&post).Error; err != nil {
      return err
    }
    
    // 更新用户帖子计数
    return tx.Model(&User{}).
      Where("id = ?", userID).
      Update("posts_count", gorm.Expr("posts_count + 1")).Error
  })
}
```

#### 获取时间线

```go
func GetFeed(userID int, page, pageSize int) ([]Post, int64, error) {
  var posts []Post
  var total int64
  
  offset := (page - 1) * pageSize
  
  // 获取关注的用户 ID
  var followingIDs []int
  db.Model(&Follow{}).
    Where("follower_id = ?", userID).
    Pluck("following_id", &followingIDs)
  
  // 添加自己的 ID
  followingIDs = append(followingIDs, userID)
  
  // 查询帖子
  query := db.Where("user_id IN ?", followingIDs).
    Preload("Author").
    Order("created_at DESC")
  
  query.Model(&Post{}).Count(&total)
  err := query.Offset(offset).Limit(pageSize).Find(&posts).Error
  
  return posts, total, err
}
```

## 中间件

### 认证中间件

```go
func AuthMiddleware() gin.HandlerFunc {
  return func(c *gin.Context) {
    authHeader := c.GetHeader("Authorization")
    if authHeader == "" {
      c.JSON(401, gin.H{"code": 401, "message": "未授权"})
      c.Abort()
      return
    }
    
    tokenString := strings.TrimPrefix(authHeader, "Bearer ")
    claims := &Claims{}
    
    token, err := jwt.ParseWithClaims(tokenString, claims, func(token *jwt.Token) (interface{}, error) {
      return []byte("your-secret-key"), nil
    })
    
    if err != nil || !token.Valid {
      c.JSON(401, gin.H{"code": 401, "message": "token 无效"})
      c.Abort()
      return
    }
    
    c.Set("userId", claims.ID)
    c.Next()
  }
}
```

  Message string      `json:"message"`
  Data    interface{} `json:"data"`
}

  c.JSON(code, ApiResponse{
    Code:    code,
    Message: message,

## 性能优化建议

1. **缓存**：使用 Redis 缓存热门数据（用户信息、热门帖子等）
2. **数据库索引**：为常查询字段添加索引
3. **异步处理**：使用消息队列处理通知、邮件等后台任务
4. **分页**：所有列表查询使用分页
5. **数据库连接池**：设置合理的连接池大小
6. **压缩**：启用 Gzip 压缩响应
